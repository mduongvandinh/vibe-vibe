# 8.4.4 Kiểm tra tự động trước cam kết—Ngăn chặn cam kết sai

.gitignore là lớp phòng thủ đầu tiên, kiểm tra pre-commit là lớp phòng thủ thứ hai—bảo hiểm kép ngăn chặn tệp nhạy cảm vào kho lưu trữ.

## Tại sao cần kiểm tra pre-commit

| Kịch bản | Vấn đề | Giải pháp |
|------|------|----------|
| Quên thêm .gitignore | Loại tệp mới được cam kết | Quét mẫu nhạy cảm tự động |
| Vòng qua .gitignore | git add -f buộc thêm | Hook chặn |
| Cam kết tệp lớn vô tình | Kho lưu trữ phình to | Kiểm tra kích thước tệp |
| Mã hóa cứng thông tin nhạy cảm | Khóa viết trong mã | Kiểm tra phù hợp mẫu |

## Sử dụng husky cấu hình pre-commit

### Cài đặt và cấu hình

```bash
# Khởi tạo husky
pnpm add -D husky
pnpm exec husky init
```

### Tạo kịch bản kiểm tra tệp nhạy cảm

Tạo `scripts/check-secrets.sh`:

```bash
#!/bin/bash

# Kiểm tra xem tệp nhạy cảm nào được đưa vào khu vực tạm thời
SENSITIVE_PATTERNS=(
  "\.env$"
  "\.env\.local$"
  "\.pem$"
  "\.key$"
  "id_rsa"
  "\.p12$"
  "credentials\.json"
  "firebase.*\.json"
)

STAGED_FILES=$(git diff --cached --name-only)
FOUND_ISSUES=0

for file in $STAGED_FILES; do
  for pattern in "${SENSITIVE_PATTERNS[@]}"; do
    if [[ $file =~ $pattern ]]; then
      echo "❌ Tệp nhạy cảm được đưa vào khu vực tạm thời: $file"
      FOUND_ISSUES=1
    fi
  done
done

if [ $FOUND_ISSUES -eq 1 ]; then
  echo ""
  echo "Vui lòng sử dụng 'git reset HEAD <file>' để xóa tệp nhạy cảm"
  echo "Nếu thực sự cần cam kết, hãy sử dụng 'git commit --no-verify'"
  exit 1
fi

echo "✅ Kiểm tra tệp nhạy cảm đã vượt qua"
```

### Cấu hình hook pre-commit

Chỉnh sửa `.husky/pre-commit`:

```bash
#!/usr/bin/env sh
. "$(dirname -- "$0")/_/husky.sh"

# Kiểm tra tệp nhạy cảm
bash scripts/check-secrets.sh

# lint-staged (nếu được cấu hình)
pnpm exec lint-staged
```

## Kiểm tra khóa mã hóa cứng

### Sử dụng gitleaks

```bash
# Cài đặt gitleaks
brew install gitleaks

# Thêm kiểm tra trong pre-commit
gitleaks protect --staged
```

### Cấu hình .gitleaks.toml

```toml
title = "Gitleaks Config"

[[rules]]
description = "AWS Access Key"
regex = '''AKIA[0-9A-Z]{16}'''
tags = ["aws", "key"]

[[rules]]
description = "OpenAI API Key"
regex = '''sk-[a-zA-Z0-9]{48}'''
tags = ["openai", "key"]

[[rules]]
description = "Generic Password"
regex = '''(?i)(password|pwd|pass)\s*[:=]\s*['"][^'"]{8,}['"]'''
tags = ["password"]
```

## Kiểm tra tệp lớn

```bash
#!/bin/bash
# scripts/check-file-size.sh

MAX_SIZE=5000000  # 5MB

STAGED_FILES=$(git diff --cached --name-only)

for file in $STAGED_FILES; do
  if [ -f "$file" ]; then
    SIZE=$(wc -c < "$file")
    if [ $SIZE -gt $MAX_SIZE ]; then
      echo "❌ Tệp quá lớn: $file ($(( SIZE / 1000000 ))MB)"
      echo "Cân nhắc sử dụng Git LFS hoặc thêm vào .gitignore"
      exit 1
    fi
  fi
done

echo "✅ Kiểm tra kích thước tệp đã vượt qua"
```

## Kiểm tra ở cấp độ CI

Ngay cả khi bỏ qua hook cục bộ, CI cũng phải có lớp phòng thủ cuối cùng.

### Kiểm tra GitHub Actions

```yaml
# .github/workflows/security.yml
name: Security Check

on: [push, pull_request]

jobs:
  secrets-scan:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Gitleaks Scan
        uses: gitleaks/gitleaks-action@v2
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

  check-ignored-files:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Check for sensitive files
        run: |
          SENSITIVE_FILES=$(find . -type f \( \
            -name ".env" \
            -o -name ".env.local" \
            -o -name "*.pem" \
            -o -name "*.key" \
            -o -name "id_rsa" \
          \) 2>/dev/null)

          if [ -n "$SENSITIVE_FILES" ]; then
            echo "::error::Found sensitive files in repository:"
            echo "$SENSITIVE_FILES"
            exit 1
          fi
```

## Cấu hình pre-commit hoàn chỉnh

```bash
#!/usr/bin/env sh
. "$(dirname -- "$0")/_/husky.sh"

echo "🔍 Chạy kiểm tra pre-commit..."

# 1. Kiểm tra tệp nhạy cảm
echo "Checking for sensitive files..."
bash scripts/check-secrets.sh || exit 1

# 2. Kiểm tra kích thước tệp
echo "Checking file sizes..."
bash scripts/check-file-size.sh || exit 1

# 3. Quét khóa (nếu cài đặt gitleaks)
if command -v gitleaks &> /dev/null; then
  echo "Scanning for secrets..."
  gitleaks protect --staged || exit 1
fi

# 4. Định dạng mã và lint
echo "Running lint-staged..."
pnpm exec lint-staged || exit 1

echo "✅ Tất cả kiểm tra pre-commit đã vượt qua!"
```
