# 12.4.3 Xây dựng phòng trò chuyện trực tuyến——Truyền thông báo thời gian thực: Quản lý phòng và phát sóng thông báo

### Nội dung chính trong một câu

Cơ chế room cho phép bạn chia nhóm người dùng, chỉ phát sóng thông báo đến các nhóm cụ thể——đây là nền tảng xây dựng ứng dụng nhiều phòng trò chuyện.

### Quản lý phòng

```typescript
// Phía máy chủ
io.on('connection', (socket) => {
  // Tham gia phòng
  socket.on('joinRoom', (roomId: string) => {
    socket.join(roomId);
    socket.to(roomId).emit('userJoined', { userId: socket.id });
    console.log(`${socket.id} tham gia phòng ${roomId}`);
  });

  // Rời khỏi phòng
  socket.on('leaveRoom', (roomId: string) => {
    socket.leave(roomId);
    socket.to(roomId).emit('userLeft', { userId: socket.id });
  });

  // Gửi thông báo đến phòng
  socket.on('roomMessage', ({ roomId, message }) => {
    io.to(roomId).emit('roomMessage', {
      userId: socket.id,
      message,
      timestamp: Date.now(),
    });
  });
});
```

### So sánh các cách phát sóng

```typescript
// Gửi cho tất cả mọi người (bao gồm cả chính mình)
io.emit('event', data);

// Gửi cho tất cả mọi người (ngoại trừ chính mình)
socket.broadcast.emit('event', data);

// Gửi cho tất cả mọi người trong phòng cụ thể
io.to('roomId').emit('event', data);

// Gửi cho phòng cụ thể (ngoại trừ chính mình)
socket.to('roomId').emit('event', data);

// Gửi cho người dùng cụ thể
io.to(socketId).emit('event', data);
```

### Thực hiện phòng trò chuyện hoàn chỉnh

#### Phía máy chủ

```typescript
interface Message {
  id: string;
  userId: string;
  username: string;
  content: string;
  timestamp: number;
}

interface Room {
  id: string;
  name: string;
  users: Set<string>;
}

const rooms = new Map<string, Room>();
const users = new Map<string, { socketId: string; username: string }>();

io.on('connection', (socket) => {
  // Đăng nhập người dùng
  socket.on('login', (username: string) => {
    users.set(socket.id, { socketId: socket.id, username });
    socket.emit('loginSuccess', { userId: socket.id });
  });

  // Lấy danh sách phòng
  socket.on('getRooms', () => {
    const roomList = Array.from(rooms.values()).map((room) => ({
      id: room.id,
      name: room.name,
      userCount: room.users.size,
    }));
    socket.emit('roomList', roomList);
  });

  // Tham gia phòng
  socket.on('joinRoom', (roomId: string) => {
    const room = rooms.get(roomId);
    if (room) {
      socket.join(roomId);
      room.users.add(socket.id);

      const user = users.get(socket.id);
      io.to(roomId).emit('userJoined', {
        userId: socket.id,
        username: user?.username,
      });
    }
  });

  // Gửi thông báo
  socket.on('sendMessage', ({ roomId, content }) => {
    const user = users.get(socket.id);
    const message: Message = {
      id: Date.now().toString(),
      userId: socket.id,
      username: user?.username || 'Nặc danh',
      content,
      timestamp: Date.now(),
    };

    io.to(roomId).emit('newMessage', message);
  });

  // Ngắt kết nối
  socket.on('disconnect', () => {
    users.delete(socket.id);
    rooms.forEach((room) => {
      if (room.users.has(socket.id)) {
        room.users.delete(socket.id);
        io.to(room.id).emit('userLeft', { userId: socket.id });
      }
    });
  });
});
```

#### Phía máy khách

```tsx
'use client';

import { useEffect, useState, useCallback } from 'react';
import { io, Socket } from 'socket.io-client';

interface Message {
  id: string;
  userId: string;
  username: string;
  content: string;
  timestamp: number;
}

export default function ChatRoom({ roomId }: { roomId: string }) {
  const [socket, setSocket] = useState<Socket | null>(null);
  const [messages, setMessages] = useState<Message[]>([]);
  const [input, setInput] = useState('');
  const [users, setUsers] = useState<string[]>([]);

  useEffect(() => {
    const newSocket = io('http://localhost:3001');
    setSocket(newSocket);

    newSocket.emit('joinRoom', roomId);

    newSocket.on('newMessage', (message: Message) => {
      setMessages((prev) => [...prev, message]);
    });

    newSocket.on('userJoined', ({ username }) => {
      setUsers((prev) => [...prev, username]);
    });

    newSocket.on('userLeft', ({ username }) => {
      setUsers((prev) => prev.filter((u) => u !== username));
    });

    return () => {
      newSocket.emit('leaveRoom', roomId);
      newSocket.close();
    };
  }, [roomId]);

  const sendMessage = useCallback(() => {
    if (socket && input.trim()) {
      socket.emit('sendMessage', { roomId, content: input });
      setInput('');
    }
  }, [socket, input, roomId]);

  return (
    <div className="flex h-screen">
      {/* Danh sách người dùng */}
      <aside className="w-48 bg-gray-100 p-4">
        <h3 className="font-bold mb-2">Người dùng trực tuyến</h3>
        {users.map((user) => (
          <div key={user}>{user}</div>
        ))}
      </aside>

      {/* Khu vực trò chuyện */}
      <main className="flex-1 flex flex-col">
        <div className="flex-1 overflow-auto p-4">
          {messages.map((msg) => (
            <div key={msg.id} className="mb-2">
              <span className="font-bold">{msg.username}: </span>
              <span>{msg.content}</span>
            </div>
          ))}
        </div>

        <div className="p-4 border-t flex gap-2">
          <input
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyPress={(e) => e.key === 'Enter' && sendMessage()}
            className="flex-1 p-2 border rounded"
            placeholder="Nhập thông báo..."
          />
          <button onClick={sendMessage} className="px-4 py-2 bg-blue-500 text-white rounded">
            Gửi
          </button>
        </div>
      </main>
    </div>
  );
}
```

### Hướng dẫn hợp tác AI

- **Ý định cốt lõi**: Cho phép AI giúp bạn thực hiện chức năng trò chuyện hoàn chỉnh.
- **Công thức xác định yêu cầu**: `"Vui lòng giúp tôi thực hiện một ứng dụng trò chuyện hỗ trợ nhiều phòng, bao gồm thông báo tham gia/rời khỏi, lịch sử thông báo và danh sách người dùng trực tuyến."`
- **Thuật ngữ chính**: `room`、`join`、`leave`、`broadcast`、`message broadcast`

### Hướng dẫn tránh hố tương tự

- **Thứ tự thông báo**: Độ trễ mạng có thể dẫn đến thông báo bị xáo trộn, hãy cân nhắc sắp xếp theo dấu thời gian.
- **Loại bỏ thông báo trùng lặp**: Khi kết nối lại, bạn có thể nhận được thông báo trùng lặp, cần loại bỏ ở phía máy khách.
- **Đồng bộ hóa trạng thái người dùng**: Sau khi ngắt kết nối và kết nối lại, cần đồng bộ hóa lại trạng thái phòng.
